1. IMU Fundamentals

An Inertial Measurement Unit (IMU) combines an accelerometer and a gyroscope (and sometimes a magnetometer) into a single chip. The K1 has an IMU mounted in its torso near the center of mass.

Sensor Measures Units Noise Type
Accelerometer Linear acceleration (includes gravity) m/s² White noise + vibration
Gyroscope Angular velocity (rotation rate) rad/s Drift + white noise

When the robot is standing still, the accelerometer reads approximately [0, 0, 9.81] m/s² — the gravity vector pointing downward in the trunk_link frame. A significant deviation from this vector indicates the robot is tilted or falling.

2. Subscribing to IMU and Joint State Topics

import rclpy
from rclpy.node import Node
from sensor_msgs.msg import Imu, JointState
import numpy as np
import matplotlib.pyplot as plt
from collections import deque
import threading

WINDOW = 200    # samples to keep (at 100 Hz → 2 seconds of data)

class SensorMonitor(Node):
    def __init__(self):
        super().__init__("sensor_monitor")
        self.accel_x = deque(maxlen=WINDOW)
        self.accel_z = deque(maxlen=WINDOW)
        self.gyro_y  = deque(maxlen=WINDOW)
        self.create_subscription(
            Imu, "/robot_interface/imu", self.imu_cb, 10)
        self.create_subscription(
            JointState, "/joint_states", self.joint_cb, 10)

    def imu_cb(self, msg: Imu):
        a = msg.linear_acceleration
        g = msg.angular_velocity
        self.accel_x.append(a.x)
        self.accel_z.append(a.z)
        self.gyro_y.append(g.y)

    def joint_cb(self, msg: JointState):
        # Log only the knee joints
        for i, name in enumerate(msg.name):
            if "knee" in name:
                pos = msg.position[i]
                vel = msg.velocity[i]
                self.get_logger().info(f"{name}: pos={pos:.3f} vel={vel:.3f}")

3. Reading Battery Level

Battery state is broadcast on /low_state as a custom Booster message. The relevant field is battery_state.percent: 

from booster_msgs.msg import LowState

class BatteryMonitor(Node):
    def __init__(self):
        super().__init__("battery_monitor")
        self.start_time = self.get_clock().now()
        self.log = []
        self.create_subscription(
            LowState, "/low_state", self.cb, 5)

    def cb(self, msg: LowState):
        pct = msg.battery_state.percent
        voltage = msg.battery_state.voltage
        elapsed = (self.get_clock().now() - self.start_time).nanoseconds / 1e9
        self.log.append((elapsed, pct, voltage))
        if pct < 20:
            self.get_logger().warn(f"LOW BATTERY: {pct:.0f}%")

4. Fall Detection Algorithm

A humanoid robot is falling when its torso tilts beyond a safe angle. We detect this by measuring the angle between the IMU's measured acceleration vector and the expected gravity vector: 

import numpy as np

GRAVITY       = np.array([0.0, 0.0, 9.81])
FALL_THRESHOLD = 35  # degrees — robot is "falling" beyond this

def tilt_angle_deg(accel: np.ndarray) -> float:
    """Returns tilt angle (degrees) between accel and gravity vectors."""
    a_norm = accel / (np.linalg.norm(accel) + 1e-9)
    g_norm = GRAVITY / np.linalg.norm(GRAVITY)
    dot = np.clip(np.dot(a_norm, g_norm), -1.0, 1.0)
    return np.degrees(np.arccos(dot))

def is_falling(accel: np.ndarray) -> bool:
    return tilt_angle_deg(accel) > FALL_THRESHOLD
Tip: Raw accelerometer data is noisy. Apply a low-pass filter (e.g., exponential moving average with α = 0.2) before computing the tilt angle to avoid false fall detections from footstep impacts. 
# Simple exponential moving average filter
alpha = 0.2
filtered = np.array([0.0, 0.0, 9.81])

def update_filter(raw_accel: np.ndarray) -> np.ndarray:
    global filtered
    filtered = alpha * raw_accel + (1 - alpha) * filtered
    return filtered

5. Plotting IMU Data with Matplotlib

import matplotlib.pyplot as plt
import matplotlib.animation as animation

fig, axes = plt.subplots(2, 1, figsize=(10, 6))
fig.suptitle("K1 IMU Live Plot")

def animate(frame, node):
    axes[0].cla(); axes[1].cla()
    axes[0].plot(node.accel_x, label="accel_x", color="r")
    axes[0].plot(node.accel_z, label="accel_z", color="b")
    axes[0].set_ylabel("Acceleration (m/s²)")
    axes[0].legend(loc="upper right")
    axes[1].plot(node.gyro_y, label="gyro_y", color="g")
    axes[1].set_ylabel("Angular Velocity (rad/s)")
    axes[1].legend(loc="upper right")
